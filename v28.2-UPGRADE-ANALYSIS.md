### 1.4 Validate logic mới — data-driven

Thay vì:
```python
if doc.line_type == "NHOM":
    if not doc.item_code: frappe.throw(...)
    if doc.calc_pattern != "LENGTH_TO_WEIGHT": frappe.throw(...)
```

Dùng:
```python
def validate_bom_item(doc, method):
    slug = frappe.get_cached_doc("AL Slug Library", doc.slug)
    category = frappe.get_cached_doc("AL Material Category", slug.category)

    # Đồng bộ category từ Slug Library
    if doc.category != slug.category:
        doc.category = slug.category

    # Validate dựa trên thuộc tính category (DATA-DRIVEN)
    if category.requires_item_code and not doc.item_code:
        frappe.throw(f"Bom Item {doc.slug}: thiếu item_code cho {category.category_name}")

    if category.requires_price_base_item and not doc.price_base_item:
        frappe.throw(f"Bom Item {doc.slug}: thiếu price_base_item cho {category.category_name}")

    if category.requires_glass_master and not doc.default_glass_master:
        frappe.throw(f"Bom Item {doc.slug}: thiếu default_glass_master cho {category.category_name}")

    if category.requires_ctx_inject_prefix and not doc.ctx_inject_prefix:
        frappe.throw(f"Bom Item {doc.slug}: thiếu ctx_inject_prefix cho {category.category_name}")

    # Validate calc_pattern thuộc danh sách được phép
    if category.allowed_calc_patterns and doc.calc_pattern:
        allowed = [row.calc_pattern for row in category.allowed_calc_patterns]
        if doc.calc_pattern not in allowed:
            frappe.throw(
                f"Bom Item {doc.slug}: calc_pattern '{doc.calc_pattern}' "
                f"không được phép cho {category.category_name}. "
                f"Các pattern được phép: {', '.join(allowed)}"
            )

    # Validate Rule mode
    if doc.item_selection_mode == "Rule":
        if not doc.rule_input_expr:
            frappe.throw(f"Bom Item {doc.slug} (mode=Rule): thiếu rule_input_expr")
```

**Kết quả:** Thêm category mới (vd: NHUA) → tạo 1 record AL Material Category với các flag đúng → validate tự động hoạt động. 0 dòng code thay đổi.

```python
def get_active_pricing_dimensions(category_code=None):
    """Lấy danh sách pricing dimensions đang active, lọc theo category."""
    filters = {"is_active": 1}
    dimensions = frappe.get_all("AL Pricing Dimension",
        filters=filters, fields=["*"], order_by="sort_order")

    if category_code:
        dimensions = [
            d for d in dimensions
            if not d.applies_to_category or category_code in d.applies_to_category
        ]
    return dimensions


def build_price_filters(item_code, price_list, price_attrs):
    """Build filters cho Item Price query từ các dimensions đang active.

    Args:
        item_code: str
        price_list: str
        price_attrs: dict — {dimension_code: value} từ user input

    Returns:
        dict — filters cho frappe.db.get_all("Item Price", filters=...)
    """
    dimensions = get_active_pricing_dimensions()
    filters = {
        "item_code": item_code,
        "price_list": price_list,
    }
    for dim in dimensions:
        fieldname = dim.custom_fieldname  # VD: custom_pd_MAU_SAC
        value = price_attrs.get(dim.dimension_code)
        if value:
            filters[fieldname] = value

    return filters
```

```python
def validate_unique_price_combo(doc, method):
```

## PHẦN 3: COST BUCKET ĐỘNG — KHÔNG CÒN VL_NHOM, VL_KINH CỨNG

### 3.1 Vấn đề

Hiện tại, Cost Template aggregate các bucket bằng tên cứng:

```python
# templates.py — FORMULAS_COST_TEMPLATE
{"name": "TONG_VL", "formula": "VL_NHOM + VL_KINH + VL_VTP + VL_PK"}

# templates.py — FORMULAS_PROJECT  
{"name": "TOTAL_VL_NHOM", "formula": "sum(v for v in all_vl_nhom)"},
{"name": "TOTAL_VL_KINH", "formula": "sum(v for v in all_vl_kinh)"},
```

Khi thêm bucket mới (VL_THEP, VL_INOX), phải sửa công thức.

### 3.2 Giải pháp

**Cách 1: Dùng AL Cost Bucket với parent_bucket**

Mỗi LEAF bucket (`VL_NHOM`, `VL_THEP`, `VL_KINH`...) được gán `parent_bucket = "TONG_VL"`. Khi gom bucket ở B5:

```python
def b5_aggregate_buckets(self):
    """Gom cost bucket động — dùng parent_bucket relationship."""
    buckets = defaultdict(float)

    # 1. Gom LEAF buckets từ row_results
    for item in self.bom_items:
        slug = item["slug"]
        bucket_code = item.get("cost_bucket")
        if bucket_code:
            line_total = self.row_results.get(f"{slug}__line_total", 0)
            buckets[bucket_code] += line_total

    # 2. Tự động aggregate lên parent buckets
    all_buckets = frappe.get_all("AL Cost Bucket",
        fields=["bucket_code", "bucket_role", "parent_bucket"])
    bucket_map = {b.bucket_code: b for b in all_buckets}

    # Duyệt từ LEAF lên AGGREGATE (theo thứ tự topo)
    changed = True
    while changed:
        changed = False
        for code, val in list(buckets.items()):
            bucket = bucket_map.get(code)
            if bucket and bucket.parent_bucket:
                parent = bucket.parent_bucket
                if buckets.get(parent, 0) != buckets.get(parent, 0):
                    pass
                buckets[parent] = buckets.get(parent, 0) + val
                changed = True

    return dict(buckets)
```

**Cách 2 (tốt hơn): Dùng Cost Template dynamic — công thức không cần liệt kê tất cả bucket**

Cost Template có 1 dòng đặc biệt:
```json
{
  "line_code": "TONG_VL",
  "calc_formula": "SUM_CHILDREN(TONG_VL)",  // Tự động sum tất cả LEAF bucket có parent=TONG_VL
  "is_subtotal": 1
}
```

FB custom function `SUM_CHILDREN` sẽ query Cost Bucket để tìm tất cả LEAF children và sum.

HOẶC đơn giản hơn: **Cost Template sinh công thức động** dựa trên cấu trúc Cost Bucket:

```python
def build_dynamic_cost_template(bom_items):
    """Sinh Cost Template động từ Cost Bucket hierarchy."""
    # 1. Thu thập tất cả LEAF buckets đang dùng trong BOM
    used_buckets = set()
    for item in bom_items:
        bucket = item.get("cost_bucket")
        if bucket:
            used_buckets.add(bucket)

    # 2. Lấy AGGREGATE buckets
    aggregates = frappe.get_all("AL Cost Bucket",
        filters={"bucket_role": "AGGREGATE"},
        fields=["bucket_code", "parent_bucket"])

    # 3. Với mỗi AGGREGATE, tìm tất cả LEAF children đang dùng
    leaf_map = defaultdict(list)
    all_buckets = frappe.get_all("AL Cost Bucket",
        fields=["bucket_code", "bucket_role", "parent_bucket"])
    bucket_by_code = {b.bucket_code: b for b in all_buckets}

    for agg in aggregates:
        children = [
            b.bucket_code for b in all_buckets
            if b.bucket_role == "LEAF"
            and b.parent_bucket == agg.bucket_code
            and b.bucket_code in used_buckets
        ]
        if children:
            leaf_map[agg.bucket_code] = children

    # 4. Sinh global_formulas
    formulas = []
    for agg_code, children in leaf_map.items():
        formula = " + ".join(children)
        formulas.append({"name": agg_code, "formula": formula})

    # 5. Các dòng còn lại của Cost Template (NC, OH, PROFIT...)
    #    vẫn dùng từ AL Cost Template đã cấu hình
    #    nhưng thay dòng TONG_VL bằng phiên bản động

    return formulas
```

**Kết quả:** Khi thêm bucket `VL_THEP` với `parent_bucket = "TONG_VL"` và `bucket_role = "LEAF"`, công thức `TONG_VL = VL_NHOM + VL_KINH + VL_VTP + VL_PK + VL_THEP` được sinh tự động. Không sửa code, không sửa Cost Template.

## PHẦN 4: LOOKUP_CALC_PATTERN — TỪ IF/ELIF SANG DATA-DRIVEN

### 4.1 Hiện trạng

```python
def lookup_calc_pattern(calc_pattern_code, width=None, height=None, weight_per_unit=None):
    w = (width or 0) / 1000.0
    h = (height or 0) / 1000.0
    tlr = weight_per_unit or 0

    if calc_pattern_code == "LENGTH_TO_WEIGHT":
        return w * tlr
    if calc_pattern_code == "AREA":
        return w * h
    if calc_pattern_code == "LENGTH_ONLY":
        return w
    if calc_pattern_code == "COUNT":
        return 1

    raise ValueError(...)
```

### 4.2 Giải pháp: AL Quantity Calc Method mở rộng

Thêm field mới vào AL Quantity Calc Method:

| Fieldname | Fieldtype | Mô tả |
|---|---|---|
| `calc_pattern_code` | Data (unique) | LENGTH_TO_WEIGHT, AREA, LENGTH_ONLY, COUNT, VOLUME... |
| `calc_pattern_name` | Data | Tên hiển thị |
| `calc_formula` | Small Text | Công thức hiển thị |
| `calc_fn` | Small Text | 🆕 **Python lambda hợp lệ (safe eval)**: `lambda w, h, tlr, **kw: (w/1000) * tlr` |
| `input_vars` | JSON | 🆕 Các biến đầu vào: `["width", "height", "weight_per_unit"]` |
| `output_unit` | Data | kg, m2, m, cai, m3... |
| `sort_order` | Int | |

**Dữ liệu mẫu mở rộng:**

| calc_pattern_code | calc_fn | input_vars | output_unit |
|---|---|---|---|
| LENGTH_TO_WEIGHT | `lambda w, h, tlr, **kw: (w/1000) * tlr` | ["width", "weight_per_unit"] | kg |
| AREA | `lambda w, h, tlr, **kw: (w/1000) * (h/1000)` | ["width", "height"] | m2 |
| LENGTH_ONLY | `lambda w, h, tlr, **kw: w/1000` | ["width"] | m |
| COUNT | `lambda w, h, tlr, **kw: 1` | [] | cai |
| VOLUME | `lambda w, h, tlr, **kw: (w/1000)*(h/1000)*(kw.get("thickness",0)/1000)` | ["width", "height", "thickness"] | m3 |

> **Bảo mật:** `calc_fn` chỉ chấp nhận lambda với restricted builtins. Dùng `ast.parse` để validate trước khi lưu. Chỉ cho phép các phép toán cơ bản (+, -, *, /, **) và biến w, h, tlr, kw.

**lookup_calc_pattern mới:**

```python
# Cache pattern functions tại module level
_PATTERN_CACHE = {}

def lookup_calc_pattern(calc_pattern_code, width=None, height=None, weight_per_unit=None, **extra_vars):
    """Tính số lượng đơn vị theo pattern — DATA-DRIVEN.

    Pattern được định nghĩa trong AL Quantity Calc Method,
    không còn if/elif chain. Thêm pattern mới = 1 record.
    """
    if calc_pattern_code not in _PATTERN_CACHE:
        pattern_doc = frappe.get_cached_doc("AL Quantity Calc Method", calc_pattern_code)
        if not pattern_doc.calc_fn:
            raise ValueError(f"Pattern '{calc_pattern_code}' chưa có calc_fn")
        # Safe eval: chỉ cho phép lambda với restricted builtins
        fn = eval(pattern_doc.calc_fn, {"__builtins__": {}}, {})
        _PATTERN_CACHE[calc_pattern_code] = fn

    fn = _PATTERN_CACHE[calc_pattern_code]
    w = (width or 0)
    h = (height or 0)
    tlr = (weight_per_unit or 0)
    return fn(w, h, tlr, **extra_vars)
```

**Lợi ích:** Thêm pattern `VOLUME` cho gỗ hoặc `SURFACE_AREA` cho sơn phủ → chỉ cần tạo 1 record AL Quantity Calc Method. 0 dòng code Python thay đổi.

## PHẦN 5: AL PRICING DIMENSION BRIDGE — CẦU NỐI VỚI VARIABLE SET

### 5.1 Vấn đề

Variable Set chứa các biến đầu vào từ user (`aluminum_color`, `aluminum_origin`...). Cần map các biến này với Pricing Dimension để khi user chọn màu → tự động biết đó là pricing dimension nào.

### 5.2 Giải pháp: Convention + Explicit Mapping

**Cách 1: Convention naming (recommended cho đơn giản)**

Variable name: `{dimension_code}_{category_code}` hoặc `pd_{dimension_code}`

VD:
- `MAU_SAC_NHOM` → dimension MAU_SAC, category NHOM
- `MAU_SAC_KINH` → dimension MAU_SAC, category KINH  
- `pd_DO_DAY` → dimension DO_DAY (chung mọi category)

**Cách 2: Explicit mapping table (cho flexibility tối đa)**

DocType: **AL Variable Dimension Mapping**

| Fieldname | Fieldtype | Mô tả |
|---|---|---|
| `variable_set` | Link → Variable Set | Bộ biến |
| `variable_name` | Data | Tên biến trong Variable Set |
| `pricing_dimension` | Link → AL Pricing Dimension | Map tới dimension |
| `material_category` | Link → AL Material Category | Áp dụng cho category nào (để trống = tất cả) |

### 5.3 Tích hợp vào Quotation flow

Khi tạo Quotation:
1. Chọn Bom Set → xác định các Bom Item + category của chúng
2. System tự động tìm các Pricing Dimension áp dụng cho các category đang dùng
3. Hiển thị form nhập giá trị cho từng dimension (từ Variable Set hoặc trực tiếp)
4. Khi B2 resolve giá, composite key tự build từ các dimension đã nhập

```python
# alumglass/patches/v28_2/migrate_line_type_to_category.py
import frappe

def execute():
    """Migrate line_type → category trong AL Slug Library và AL Bom Item."""

    # 1. Đảm bảo AL Material Category đã có dữ liệu
    mappings = {
        "ALUMINUM": "ALUMINUM",
        "GLASS": "GLASS",
        "SUPPLIES": "SUPPLIES",
        "ACCESSORY": "ACCESSORY",
        "ACCESSORY": "ACCESSORY",
    }

    # 2. Cập nhật AL Slug Library
    slugs = frappe.get_all("AL Slug Library", fields=["name", "line_type"])
    for s in slugs:
        cat_code = mappings.get(s.line_type)
        if cat_code:
            frappe.db.set_value("AL Slug Library", s.name, "category", cat_code)

    # 3. Cập nhật AL Bom Item
    items = frappe.get_all("AL Bom Item", fields=["name", "slug"])
    for item in items:
        slug = frappe.get_cached_doc("AL Slug Library", item.slug)
        if slug.category:
            frappe.db.set_value("AL Bom Item", item.name, "category", slug.category)

    frappe.db.commit()
```

## PHẦN 8: KẾT LUẬN

Với các thay đổi trên, AlumGlass v28.2 đạt được:

1. **Không hardcode loại vật tư** — Thêm ngành mới (thép, inox, nhựa, gỗ...) = 1 record AL Material Category, 0 dòng code
2. **Không hardcode đặc tính giá** — Thêm đặc tính mới (độ bóng, tiêu chuẩn kỹ thuật...) = 1 record AL Pricing Dimension, tự động tạo custom field + validate + lookup

4. **Không hardcode calc pattern** — Pattern mới (VOLUME, SURFACE_AREA...) = 1 record AL Quantity Calc Method với lambda
5. **Không hardcode validate** — Dùng flag trên category (requires_*, has_*) thay vì if category_code == "..."
6. **Tận dụng tối đa ERPNext Core** — Custom Field API, frappe.get_cached_doc, Link field với cache, DocType permissions
7. **Tận dụng tối đa Formula Builder** — BatchBindingResolver cho data source, EngineConfig cho cost template, DAG cho cross-row reference
